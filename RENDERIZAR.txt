Comandos para renderizar os arquivos XML e gerar o livro LFS em vários formatos


1 - Para a versão SystemV do livro:


cd lfs-en; git checkout -b 13.0 --track origin/13.0
cd fr-lfs-git/lfs/lfsgen-pt_BR
( time LC_ALL=C make -j1 V=1 L=pt_BR REV=sysv BASEDIR=/home/jamenson/sysv nochunks 2>&1 | tee /home/jamenson/genhtml-nochunks-sysv.log.txt && exit "${PIPESTATUS}" )
mv -v LFS-BOOK.html LFS-SYSV-BOOK-13.0-pt_BR-NOCHUNKS.html
lynx -dump LFS-SYSV-BOOK-13.0-pt_BR-NOCHUNKS.html > LFS-SYSV-BOOK-13.0-pt_BR-NOCHUNKS.txt
cd ..
rm -rf 1/
mkdir -v /tmp/1; cd ${_}; ls -al
make BASEDIR=/tmp/1 pdf
make BASEDIR=/tmp/1
tar --create --file=/home/jamenson/repos/LFS/LFS-SYSV-BOOK-13.0-pt_BR-HTML.tar --wildcards .
xz --verbose /home/jamenson/repos/LFS/LFS--SYSV-BOOK-13.0-pt_BR-HTML.tar
